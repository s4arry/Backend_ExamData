บท 01 เริ่มใช้ชีทจากตรงนี้

ชีทนี้ใช้ทำข้อสอบเขียน Backend ด้วย Node.js, Express และ MySQL แบบเปิดหนังสือและไม่ใช้อินเทอร์เน็ต เริ่มจากอ่านโจทย์ให้รู้ว่าต้องรับอะไร คืนอะไร แล้วแก้ทีละไฟล์ โค้ดเต็มอยู่ภาคท้ายและในโฟลเดอร์ practice-project ของชุดไฟล์

เวลาสอบให้ยึด spec และ template ที่อาจารย์แจกเป็นหลัก ชุดโค้ดประกอบเป็นเฉลยสำหรับแบบฝึก Mobile Distributor ในชีทนี้ ชื่อตาราง field รูปแบบ JSON และเงื่อนไขของข้อสอบจริงอาจต่างออกไป

  2. อ่านโจทย์และ SQL แล้วกรอกตารางในบท 03 ก่อนเริ่มคัดลอกโค้ด
  3. ตั้งรหัสนักศึกษา ชื่อ secretKey และการเชื่อมต่อ MySQL ให้ตรงโจทย์
  4. รัน entry file ที่แก้จริง เช่น node src/app.js แล้วเปิด student-info เพื่อเช็กเส้นทาง
  5. ทำ GET ทีละรายการก่อน แล้วทำ list และ pagination จากนั้น POST, PUT และ DELETE
  6. ทดสอบทั้งกรณีสำเร็จและกรณีผิด พร้อมตรวจชื่อ field ชนิดข้อมูล และสถานะ HTTP

บท 02 เตรียมให้ใช้งานได้ก่อนตัดอินเทอร์เน็ต

ตัวชีท PDF และ HTML เปิดออฟไลน์ได้ แต่โปรเจกต์ Express ต้องมี Node.js แพ็กเกจ npm และฐานข้อมูลพร้อมอยู่ในเครื่องก่อน ZIP ต้นฉบับที่แนบมาไม่มี node_modules จึงยังไม่ใช่โปรเจกต์ที่แตกแล้วรันได้ทันที

ทำก่อนวันสอบหรือก่อนตัดอินเทอร์เน็ต

  1. เตรียม Node.js, VS Code และ MySQL ที่ใช้จริง เปิดโปรแกรมให้ได้ เก็บ SQL และคำสั่งจากอาจารย์ไว้ในเครื่อง
  2. แตก ZIP และเปิดโฟลเดอร์ชั้นที่มี package.json ไม่ใช่ชั้นนอกที่มีแต่ชื่อโฟลเดอร์ซ้อนกัน
  3. ตรวจ node -v และ npm -v ถ้าไม่รู้จักคำสั่ง ให้แก้การติดตั้งหรือ PATH ก่อนวันสอบ
  4. ขณะยังมีอินเทอร์เน็ต รัน npm ci ในโปรเจกต์ที่มี package-lock.json ตรงกับ package.json ถ้าไม่มี lockfile ใช้ npm install ตามสภาพโปรเจกต์
  5. เก็บทั้งโปรเจกต์รวม node_modules ไว้ตามกติกาที่อนุญาต แล้วตั้ง DB และทดสอบ endpoint จริง
  6. ปิดอินเทอร์เน็ตและทดลองอีกครั้ง เปิด HTML เริ่ม Node เรียก localhost อ่านข้อมูล และสร้างข้อมูลในฐานฝึกให้ได้
Terminal ก่อนตัดอินเทอร์เน็ตเท่านั้น
node -v
npm -v
npm ci

อย่ารัน npm ci ซ้ำเพื่อซ่อมโปรเจกต์ระหว่างออฟไลน์บนสำเนาที่พร้อมใช้เพียงชุดเดียว คำสั่งนี้ลบ node_modules เดิมก่อนติดตั้งใหม่ และอาจติดตั้งกลับไม่ได้เมื่อไม่มีแพ็กเกจใน cache ครบ npm install --offline ก็ต้องมี cache ครบเช่นกัน

ฐานและตารางนำเข้า SQL ที่กำหนดแล้ว คำสั่ง SELECT อ่านแถวได้
ไฟล์โจทย์และตัวอย่างดาวน์โหลดเก็บไว้จริง เปิดได้โดยไม่พึ่งลิงก์เว็บไซต์

localhost และ 127.0.0.1 หมายถึงเครื่องตัวเอง จึงเรียก API ในเครื่องได้แม้ไม่มีอินเทอร์เน็ต ถ้าฐานข้อมูลอยู่เครื่องอื่น ต้องใช้เครือข่ายห้องสอบที่ได้รับอนุญาตตามใบงาน การพิมพ์ว่า Server running เพียงอย่างเดียวยังไม่ยืนยันว่าเชื่อม MySQL สำเร็จ ต้องลอง GET ที่อ่านฐานด้วย

import { readId } from '../validators/id-validator.js'

export async function remove(id) {
  id = readId(id)
  const brand = await repo.findOne(id)
  if (!brand) {
    throw new AppError('Brand not found', 404, 'BRAND_NOT_FOUND')
  }
  try {
    await repo.remove(id)
  } catch (error) {
    if (error.code === 'ER_ROW_IS_REFERENCED_2') {
      throw new AppError(
        'Brand is still used by products',
        409, 'RESOURCE_IN_USE'
      )
    }
    throw error
  }
}

query.sortBy เอามาต่อ SQL ตรง ๆ ไม่ได้ และ ORDER BY ? ไม่ได้แทนชื่อคอลัมน์ตามที่ต้องการ เลือกชื่อ SQL จากรายการที่โค้ดอนุญาต หากโจทย์สั่งค่าผิดต้อง 400 ให้ปฏิเสธแทนการใช้ default

นับจำนวนสินค้าของแต่ละ brand

SQL สำหรับ aggregate
SELECT b.id, b.name, COUNT(p.id) AS productCount
FROM brands b
LEFT JOIN products p ON p.brand_id = b.id
GROUP BY b.id, b.name
ORDER BY b.id;

LEFT JOIN ทำให้ brand ที่ไม่มีสินค้าอยู่ในผลด้วย ใช้ COUNT(p.id) เพื่อให้ได้ 0 สำหรับ brand ว่าง ถ้าใช้ COUNT(*) อาจนับแถวด้านซ้ายเป็น 1 ชื่อ productCount และ DTO ต้องตามโจทย์

export function parsePagination(query = {}) {
  const rawPage = Number(query.page)
  const rawLimit = Number(query.limit)
  const page = Number.isSafeInteger(rawPage) && rawPage > 0 ? rawPage : 1
  const requested = Number.isSafeInteger(rawLimit) && rawLimit > 0 ? rawLimit : 10
  const limit = Math.min(requested, 100)
  const offset = (page - 1) * limit
  // กันค่า offset เกินช่วงตัวเลขที่ JavaScript แทนได้แน่นอน
  if (!Number.isSafeInteger(offset)) return { page: 1, limit, offset: 0 }
  return { page, limit, offset }
}

export function paginationInfo(total, { page, limit }) {
  const totalPages = Math.ceil(total / limit)
  return {
    total,
    page,
    limit,
    totalPages,
    hasNextPage: page < totalPages,
    hasPrevPage: page > 1
  }
}

export function errorHandler(err, req, res, next) {
  if (res.headersSent) return next(err)

  let status = err.statusCode || 500
  let code = err.code || 'INTERNAL_ERROR'
  let message = err.message || 'Unexpected server error'

  if (err.code === 'ER_DUP_ENTRY') {
    status = 409
    code = 'DUPLICATE_KEY'
  } else if (err.code === 'ER_NO_REFERENCED_ROW_2') {
    status = 400
    code = 'INVALID_REFERENCE'
  } else if (err.code === 'ER_ROW_IS_REFERENCED_2') {
    status = 409
    code = 'RESOURCE_IN_USE'
  } else if (err.type === 'entity.parse.failed') {
    status = 400
    code = 'INVALID_JSON'
    message = 'Request body must be valid JSON'
  }

  if (status === 500) console.error(err)

  return res.status(status).json({
    success: false,
    error: { code, message },
    timestamp: new Date().toISOString()
  })
}

function text(value, field, maximum, required = false) {
  if (value === undefined || value === null) {
    if (required) invalid("Field '" + field + "' is required")
    return null
  }
  if (typeof value !== 'string') invalid(field + ' must be a string')
  const result = value.trim()
  if (required && result.length === 0) invalid("Field '" + field + "' is required")
  if (result.length > maximum) invalid(field + ' is too long')
  return result
}

  if (value === undefined || value === null) return null
  if (typeof value !== 'number' || !Number.isSafeInteger(value) || value < 0 || value > MAX_INT) {
    invalid(field + ' must be a non-negative integer or null')

export function validDate(value) {
  if (typeof value !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(value)) return false
  const [year, month, day] = value.split('-').map(Number)
  if (year < 1000 || year > 9999) return false
  const date = new Date(Date.UTC(year, month - 1, day))
  return date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 && date.getUTCDate() === day
}

export function normalizeAndValidate(body) {

function buildFilter({ search, brandId }) {
  const conditions = []
  const params = []
  if (search) {
    conditions.push('(p.name LIKE ? OR p.sku LIKE ?)')
    const keyword = '%' + search + '%'
    params.push(keyword, keyword)
  }
  if (brandId !== undefined) {
    conditions.push('p.brand_id = ?')
    params.push(brandId)
  }
  const where = conditions.length > 0 ? ' WHERE ' + conditions.join(' AND ') : ''
  return { where, params }
}

}

  }
  if (query.brandId !== undefined && typeof query.brandId !== 'string') {
